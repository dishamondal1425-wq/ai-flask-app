name: ML CI/CD Pipeline

 

on:
  push:
    branches:
      - main

 

jobs:
  build:

 

    runs-on: windows-latest

 

    steps:

 

    - name: Checkout Code
      uses: actions/checkout@v4

 

    - name: Set up Python
      uses: actions/setup-python@v5
      with:
        python-version: 3.11

 

    - name: Install Dependencies
      run: |
        pip install -r requirements.txt

 

    - name: Train Model
      run: |
        python train.py

 

    - name: Run Tests
      run: |
        pytest
